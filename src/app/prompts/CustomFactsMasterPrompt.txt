```javascript
(async () => {
  // ====== SIRF YE DATA BADLO ======
  const DATA = {
    title: "TITLE_HERE",
    description: "DESCRIPTION_HERE",
    tags: ["tag one", "tag two"],
    notMadeForKids: true,
  };
  // ================================

  const TAG = "[YT-FILL]";
  const sleep = (ms) => new Promise((r) => setTimeout(r, ms));
  const ok = (m) => console.log("%c" + TAG + " OK   " + m, "color:#22c55e;font-weight:bold");
  const bad = (m) => console.warn(TAG + " FAIL " + m);
  const $all = (sel) => Array.from(document.querySelectorAll(sel));
  const visible = (el) => !!(el && (el.offsetWidth || el.offsetHeight || el.getClientRects().length));
  const textOf = (el) => (el && (el.innerText || el.textContent) || "").trim();

  async function waitFor(fn, timeout = 20000, step = 250) {
    const t0 = Date.now();
    while (Date.now() - t0 < timeout) {
      try {
        const v = fn();
        if (v) return v;
      } catch (e) {}
      await sleep(step);
    }
    return null;
  }

  async function setEditable(el, text) {
    el.scrollIntoView({ block: "center" });
    el.focus();
    await sleep(150);
    const sel = window.getSelection();
    const range = document.createRange();
    range.selectNodeContents(el);
    sel.removeAllRanges();
    sel.addRange(range);
    document.execCommand("delete", false, null);
    const lines = String(text).split("\n");
    for (let i = 0; i < lines.length; i++) {
      if (i > 0) document.execCommand("insertLineBreak", false, null);
      if (lines[i]) document.execCommand("insertText", false, lines[i]);
    }
    el.dispatchEvent(new Event("input", { bubbles: true }));
    el.dispatchEvent(new Event("change", { bubbles: true }));
    await sleep(300);
    return textOf(el).length > 0;
  }

  // ---- limits (YouTube: title 100, description 5000, tags 500 total) ----
  let title = String(DATA.title || "").trim();
  if (title.length > 100) {
    title = title.slice(0, 100);
    bad("Title 100 chars se lamba tha, kaat diya.");
  }
  const description = String(DATA.description || "").slice(0, 5000);
  const tags = [];
  let tagChars = 0;
  for (const t of DATA.tags || []) {
    const clean = String(t).replace(/[#<>,]/g, "").trim();
    if (!clean) continue;
    if (tagChars + clean.length + 1 > 500) break;
    tags.push(clean);
    tagChars += clean.length + 1;
  }

  window.__YT_DATA = { title, description, tags: tags.join(", ") };
  const summary = {};

  // ---- wait for Details page ----
  const boxes = () =>
    $all('#textbox[contenteditable="true"], div[contenteditable="true"]').filter(visible);
  const ready = await waitFor(() => boxes().length >= 1, 30000);
  if (!ready) {
    bad("Details page nahi mila. Pehle video upload karo, jab Title/Description box dikh jayein tab script chalao.");
    return;
  }

  // ---- title ----
  const byLabel = (re) => boxes().find((b) => re.test(b.getAttribute("aria-label") || ""));
  const titleBox =
    document.querySelector("#title-textarea #textbox") || byLabel(/title/i) || boxes()[0];
  if (titleBox && (await setEditable(titleBox, title))) {
    ok("Title set (" + title.length + " chars)");
    summary.title = "OK";
  } else {
    bad("Title box nahi mila. Manual: copy(__YT_DATA.title)");
    summary.title = "FAIL";
  }

  // ---- description ----
  const descBox =
    document.querySelector("#description-textarea #textbox") ||
    byLabel(/description|tell viewers/i) ||
    boxes()[1];
  if (descBox && descBox !== titleBox && (await setEditable(descBox, description))) {
    ok("Description set (" + description.length + " chars)");
    summary.description = "OK";
  } else {
    bad("Description box nahi mila. Manual: copy(__YT_DATA.description)");
    summary.description = "FAIL";
  }

  // ---- audience: not made for kids ----
  if (DATA.notMadeForKids) {
    const radio =
      document.querySelector('[name="VIDEO_MADE_FOR_KIDS_NOT_MFK"]') ||
      $all("tp-yt-paper-radio-button, [role='radio']").find((e) =>
        /not made for kids/i.test(textOf(e))
      );
    if (radio) {
      radio.scrollIntoView({ block: "center" });
      radio.click();
      await sleep(300);
      ok("Audience: not made for kids");
      summary.audience = "OK";
    } else {
      bad("Audience radio nahi mila. Manual: 'No, it's not made for kids' select karo.");
      summary.audience = "FAIL";
    }
  }

  // ---- tags (Show more ke andar) ----
  if (tags.length) {
    const findTagInput = () =>
      $all(
        "#tags-container input, ytcp-form-input-container#tags-container input, input[aria-label*='tag' i]"
      ).find(visible);
    let tagInput = findTagInput();
    if (!tagInput) {
      const more =
        document.querySelector("#toggle-button") ||
        $all("ytcp-button, button, tp-yt-paper-button").find((e) =>
          /^show more$/i.test(textOf(e))
        );
      if (more) {
        more.scrollIntoView({ block: "center" });
        more.click();
        tagInput = await waitFor(findTagInput, 6000);
      }
    }
    if (tagInput) {
      tagInput.scrollIntoView({ block: "center" });
      tagInput.focus();
      const setter = Object.getOwnPropertyDescriptor(HTMLInputElement.prototype, "value").set;
      setter.call(tagInput, tags.join(","));
      tagInput.dispatchEvent(new Event("input", { bubbles: true }));
      await sleep(200);
      tagInput.dispatchEvent(
        new KeyboardEvent("keydown", { key: "Enter", code: "Enter", keyCode: 13, bubbles: true })
      );
      tagInput.dispatchEvent(new Event("change", { bubbles: true }));
      tagInput.blur();
      await sleep(300);
      ok("Tags bheje (" + tags.length + "). Ek baar aankh se verify karo.");
      summary.tags = "SENT (verify)";
    } else {
      bad("Tags box nahi mila. Manual: Show more kholo, phir copy(__YT_DATA.tags) paste karo.");
      summary.tags = "FAIL";
    }
  }

  console.table(summary);
  console.log(
    "%c" + TAG + " Done. Script publish nahi karti. Thumbnail, AI-disclosure aur Next/Publish tum khud karo.",
    "color:#38bdf8;font-weight:bold"
  );
})();
```
