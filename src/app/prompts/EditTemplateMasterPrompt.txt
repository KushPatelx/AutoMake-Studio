import React from "react";
import { AbsoluteFill, Img, interpolate, useCurrentFrame } from "remotion";
import { CLAMP, ProgressBar } from "../shared";
import { EditProps, EditSceneProps, EditVideo, beatPulseAt } from "../editShared";

const Scene: React.FC<EditSceneProps> = ({ src, seg, index, totalFrames, beatFrames, brand }) => {
  const frame = useCurrentFrame();
  const g = seg.start + frame;
  const pulse = beatPulseAt(g, beatFrames);
  const zoom = interpolate(frame, [0, seg.duration], [1.05, 1.15], CLAMP) + pulse * 0.03;
  const flash = interpolate(frame, [0, 6], [index === 0 ? 0 : 0.7, 0], CLAMP);
  return (
    <AbsoluteFill style={{ backgroundColor: "#000", overflow: "hidden" }}>
      <Img src={src} style={{ width: "100%", height: "100%", objectFit: "cover", transform: `scale(${zoom})` }} />
      <AbsoluteFill style={{ backgroundColor: "#fff", opacity: flash }} />
      <ProgressBar progress={(g / totalFrames) * 100} accent={brand.color} height={5} />
    </AbsoluteFill>
  );
};

export const MyEdit: React.FC<EditProps> = (props) => (
  <EditVideo {...props} bg="#000" overlap={0} Scene={Scene} />
);
